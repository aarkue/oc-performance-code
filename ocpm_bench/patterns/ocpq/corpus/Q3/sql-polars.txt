WITH offer_counts AS (
    SELECT
        E2O.ocel_event_id,
        COUNT(*) AS cnt
    FROM
        event_object AS E2O
        INNER JOIN object_Offer AS A1 ON E2O.ocel_object_id = A1.ocel_id
    GROUP BY
        E2O.ocel_event_id
)
SELECT
    E1.ocel_id,
    (COALESCE(OC.cnt, 0) = 1) AS satisfied
FROM
    event_O_Returned AS E1
    LEFT JOIN offer_counts AS OC ON OC.ocel_event_id = E1.ocel_id
