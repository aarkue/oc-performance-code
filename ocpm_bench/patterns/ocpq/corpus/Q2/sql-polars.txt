WITH returned_counts AS (
    SELECT
        XE2O.ocel_object_id,
        COUNT(*) AS cnt
    FROM
        event_O_Returned AS X
        INNER JOIN event_object AS XE2O ON XE2O.ocel_event_id = X.ocel_id
    GROUP BY
        XE2O.ocel_object_id
)
SELECT
    A1.ocel_id AS offer_id,
    E1.ocel_id AS event_id,
    (COALESCE(RC.cnt, 0) >= 1) AS satisfied
FROM
    object_Offer AS A1
    INNER JOIN event_object AS E2O ON E2O.ocel_object_id = A1.ocel_id
    INNER JOIN event_O_Created AS E1 ON E2O.ocel_event_id = E1.ocel_id
    LEFT JOIN returned_counts AS RC ON RC.ocel_object_id = A1.ocel_id
