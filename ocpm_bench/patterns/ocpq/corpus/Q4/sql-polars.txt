WITH AcceptedOffers AS (
    SELECT DISTINCT
        O2O.ocel_source_id,
        E2.acc_evt_id,
        E2.acc_time
    FROM
        object_object AS O2O
        INNER JOIN (SELECT ocel_id AS offer_id FROM object_Offer) AS A2
            ON O2O.ocel_target_id = A2.offer_id
        INNER JOIN event_object AS E2O2 ON E2O2.ocel_object_id = A2.offer_id
        INNER JOIN (SELECT ocel_id AS acc_evt_id, ocel_time AS acc_time FROM event_O_Accepted) AS E2
            ON E2O2.ocel_event_id = E2.acc_evt_id
),
HasQualified AS (
    SELECT DISTINCT
        A1.app_id AS hq_app_id,
        E1.app_evt_id AS hq_evt_id
    FROM
        (SELECT ocel_id AS app_id FROM object_Application) AS A1
        INNER JOIN event_object AS E2O ON E2O.ocel_object_id = A1.app_id
        INNER JOIN (SELECT ocel_id AS app_evt_id, ocel_time AS app_evt_time FROM event_A_Accepted) AS E1
            ON E2O.ocel_event_id = E1.app_evt_id
        INNER JOIN AcceptedOffers AS AO ON AO.ocel_source_id = A1.app_id
    WHERE E1.app_evt_time <= AO.acc_time
)
SELECT
    A1.app_id AS application_id,
    E1.app_evt_id AS event_id,
    (HQ.hq_app_id IS NOT NULL) AS satisfied
FROM
    (SELECT ocel_id AS app_id FROM object_Application) AS A1
    INNER JOIN event_object AS E2O ON E2O.ocel_object_id = A1.app_id
    INNER JOIN (SELECT ocel_id AS app_evt_id, ocel_time AS app_evt_time FROM event_A_Accepted) AS E1
        ON E2O.ocel_event_id = E1.app_evt_id
    LEFT JOIN HasQualified AS HQ ON HQ.hq_app_id = A1.app_id AND HQ.hq_evt_id = E1.app_evt_id
