SELECT MAX(duration_seconds) AS max_duration
FROM (
    SELECT E2.acc_time - E1.created_time AS duration_seconds
    FROM event_object AS E2O
        INNER JOIN (SELECT ocel_id AS offer_id FROM object_Offer) AS A1
            ON E2O.ocel_object_id = A1.offer_id
        INNER JOIN (SELECT ocel_id AS created_id, ocel_time AS created_time FROM event_O_Created) AS E1
            ON E2O.ocel_event_id = E1.created_id
        INNER JOIN event_object AS E2O2 ON E2O2.ocel_object_id = A1.offer_id
        INNER JOIN (SELECT ocel_id AS acc_id, ocel_time AS acc_time FROM event_O_Accepted) AS E2
            ON E2O2.ocel_event_id = E2.acc_id
)
