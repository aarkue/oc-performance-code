SELECT DISTINCT
    EO1.ocel_object_id AS offer_id,
    E1.ocel_id AS o_created_id,
    E2.ocel_id AS o_cancelled_id
FROM event_object_O_Created__Offer AS EO1
    JOIN event_O_Created AS E1 ON E1.ocel_id = EO1.ocel_event_id
    JOIN event_object_O_Cancelled__Offer AS EO2 ON EO2.ocel_object_id = EO1.ocel_object_id
    JOIN event_O_Cancelled AS E2 ON E2.ocel_id = EO2.ocel_event_id
WHERE E1.ocel_time <= E2.ocel_time
